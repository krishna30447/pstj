import java.util.HashMap;
import java.util.Map;

class UndergroundSystem {

   
    private Map<Integer, CheckIn> checkIns;

    
    private Map<String, RouteData> routes;

    public UndergroundSystem() {
        checkIns = new HashMap<>();
        routes = new HashMap<>();
    }

    public void checkIn(int id, String stationName, int t) {
        checkIns.put(id, new CheckIn(stationName, t));
    }

    public void checkOut(int id, String stationName, int t) {
        CheckIn checkIn = checkIns.get(id);

        String startStation = checkIn.stationName;
        int travelTime = t - checkIn.time;

        String routeKey = startStation + "->" + stationName;

        RouteData data = routes.getOrDefault(routeKey, new RouteData());

        data.totalTime += travelTime;
        data.tripCount++;

        routes.put(routeKey, data);

        
        checkIns.remove(id);
    }

    public double getAverageTime(String startStation, String endStation) {
        String routeKey = startStation + "->" + endStation;

        RouteData data = routes.get(routeKey);

        return (double) data.totalTime / data.tripCount;
    }

    
    private static class CheckIn {
        String stationName;
        int time;

        CheckIn(String stationName, int time) {
            this.stationName = stationName;
            this.time = time;
        }
    }

    
    private static class RouteData {
        int totalTime = 0;
        int tripCount = 0;
    }
}

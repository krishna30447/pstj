import java.io.*;

public class Solution {

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(
                new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine().trim());
        String s = br.readLine().trim();

        if (n == 1) {
            System.out.println(1);
            return;
        }

        
        String t = s + s;
        int m = t.length();

        
        int[] d1 = new int[m];
        int[] d2 = new int[m];

        
        int l = 0, r = -1;

        for (int i = 0; i < m; i++) {
            int k = (i > r) ? 1 : Math.min(d1[l + r - i], r - i + 1);

            while (i - k >= 0 && i + k < m &&
                    t.charAt(i - k) == t.charAt(i + k)) {
                k++;
            }

            d1[i] = k--;

            if (i + k > r) {
                l = i - k;
                r = i + k;
            }
        }

        
        l = 0;
        r = -1;

        for (int i = 0; i < m; i++) {
            int k = (i > r) ? 0 : Math.min(d2[l + r - i + 1], r - i + 1);

            while (i - k - 1 >= 0 && i + k < m &&
                    t.charAt(i - k - 1) == t.charAt(i + k)) {
                k++;
            }

            d2[i] = k--;

            if (i + k > r) {
                l = i - k - 1;
                r = i + k;
            }
        }

       

        for (int start = 0; start < n; start++) {

            int end = start + n - 1;
            int answer = 1;

            
            for (int center = start; center <= end; center++) {
                int radius = d1[center];

                int maxRadius = Math.min(
                        radius,
                        Math.min(center - start + 1,
                                end - center + 1)
                );

                answer = Math.max(answer, 2 * maxRadius - 1);
            }

            
            for (int center = start; center <= end; center++) {
                int radius = d2[center];

                int maxRadius = Math.min(
                        radius,
                        Math.min(center - start,
                                end - center + 1)
                );

                answer = Math.max(answer, 2 * maxRadius);
            }

            System.out.println(answer);
        }
    }
}
